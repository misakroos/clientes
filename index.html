begin; -- MISAKROOS CRM v18 -- Renovación configurable por servicio: -- days = cantidad exacta de días -- monthly_same_day = mismo día del mes siguiente -- manual = vencimiento elegido manualmente alter table public.services add column if not exists renewal_type text not null default 'days'; alter table public.services add column if not exists renewal_days integer not null default 30; alter table public.services drop constraint if exists services_renewal_type_check; alter table public.services add constraint services_renewal_type_check check (renewal_type in ('days','monthly_same_day','manual')); alter table public.services drop constraint if exists services_renewal_days_check; alter table public.services add constraint services_renewal_days_check check (renewal_days > 0); -- Conservamos la duración que ya tenía cada servicio. update public.services set renewal_days = coalesce(default_duration_days,30) where renewal_type = 'days'; create or replace function public.calculate_service_end_date( p_service_id uuid, p_start_date date ) returns date language plpgsql stable set search_path = '' as $$ declare v_type text; v_days integer; begin if p_start_date is null then return null; end if; select coalesce(s.renewal_type,'days'), coalesce(s.renewal_days,s.default_duration_days,30) into v_type,v_days from public.services s where s.id=p_service_id; if not found then return p_start_date + 30; end if; if v_type='manual' then return null; elsif v_type='monthly_same_day' then return (p_start_date + interval '1 month')::date; else return p_start_date + greatest(v_days,1); end if; end; $$; create or replace function public.apply_sale_item_renewal_rule() returns trigger language plpgsql set search_path = '' as $$ declare v_type text; v_end date; begin select coalesce(s.renewal_type,'days') into v_type from public.services s where s.id=new.service_id; -- En modo manual respetamos exactamente el vencimiento elegido por el usuario. if v_type='manual' then return new; end if; v_end=public.calculate_service_end_date(new.service_id,new.start_date); if v_end is not null then new.end_date=v_end; end if; return new; end; $$; drop trigger if exists zz_sale_items_apply_renewal_rule on public.sale_items; create trigger zz_sale_items_apply_renewal_rule before insert or update of start_date, service_id on public.sale_items for each row execute function public.apply_sale_item_renewal_rule(); notify pgrst, 'reload schema'; commit;
